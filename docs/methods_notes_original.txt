#!/usr/bin/env python3
import os
import sys
import csv
import json
import re
import subprocess


# ================== Step0: 工具路径 ==================
GUIDESCAN_BIN = "~/NtExpDB/NtExpAtlas/guidescan/bin/guidescan"
GENERATE_KMERS = "~/NtExpDB/NtExpAtlas/guidescan/scripts/generate_kmers.py"


# ================== Step1: sgRNA过滤 ==================
poly_pattern = re.compile(r'(A{4,}|T{4,}|C{4,}|G{4,})')
dinuc_pattern = re.compile(r'((AT|TA|CG|GC){4,})')
tttt_pattern = re.compile(r'T{4}')


def gc_content(seq):
    return sum(1 for c in seq if c in "GC") / len(seq)


def seed_gc_ok(seq):
    seed = seq[-12:]
    gc = (seed.count('G') + seed.count('C')) / 12
    return 0.2 <= gc <= 0.8


def pass_filter(seq):
    seq = seq.upper()
    if len(seq) != 20: return False
    if 'N' in seq: return False
    if not (0.2 <= gc_content(seq) <= 0.8): return False
    if poly_pattern.search(seq): return False
    if dinuc_pattern.search(seq): return False
    if tttt_pattern.search(seq): return False
    if not seed_gc_ok(seq): return False
    return True


def filter_sgrna(infile, outfile):
    with open(infile) as f, open(outfile, "w") as out:
        out.write("id,sequence,pam,chromosome,position,sense\n")
        next(f)
        for line in f:
            seq = line.strip().split(',')[1]
            if pass_filter(seq):
                out.write(line)


# ================== Step2: off-target统计 ==================
def safe_int(x):
    try:
        return int(x)
    except:
        return 0


def summarize_offtarget(infile, outfile):
    current_id = None
    mm0 = mm1 = bulge = 0
    current_seq = ""
    spec = 0

    with open(infile) as f, open(outfile, "w") as out:
        next(f)
        out.write("id,sgRNA,mm0_count,mm1_count,bulge_count,specificity\n")

        for line in f:
            parts = line.strip().split(',')
            if "NA" in parts:
                continue
            if "specificity" in parts:
                continue

            sg_id = parts[0]
            sgRNA = parts[1]
            dist = safe_int(parts[5])
            rna_b = safe_int(parts[7])
            dna_b = safe_int(parts[8])
            spec = parts[9]

            if current_id is None:
                current_id = sg_id
                current_seq = sgRNA
                current_spec = spec

            if sg_id != current_id:
                out.write(f"{current_id},{current_seq},{mm0},{mm1},{bulge},{current_spec}\n")
                current_id = sg_id
                current_seq = sgRNA
                current_spec = spec
                mm0 = mm1 = bulge = 0

            if dist == 0:
                mm0 += 1
            elif dist == 1:
                mm1 += 1

            if rna_b > 0 or dna_b > 0:
                bulge += 1

        if current_id:
            out.write(f"{current_id},{current_seq},{mm0},{mm1},{bulge},{spec}\n")


# ================== Step3: FASTA读取 ==================
def read_fasta_multi(fasta):
    seqs = {}
    current = None
    with open(fasta) as f:
        for line in f:
            if line.startswith(">"):
                current = line[1:].split()[0]
                seqs[current] = ""
            else:
                seqs[current] += line.strip()
    return seqs


def parse_id(s):
    gene = s.replace("sgRNA", "").split(":")[0]
    pos = int(s.split(":")[1])
    strand = s.split(":")[2]
    return gene, pos, strand


# ================== Step4: JSON + CSV ==================
def build_outputs(fasta, sgrna_csv, out_json, out_csv, pam):
    seqs = read_fasta_multi(fasta)
    result = {}
    table = []

    with open(sgrna_csv) as f:
        reader = csv.DictReader(f)

        for i, row in enumerate(reader):
            sg_id = row["id"]
            seq = row["sgRNA"]

            gene, pos, strand = parse_id(sg_id)
            if gene not in seqs:
                continue

            if gene not in result:
                result[gene] = {"gDNA": seqs[gene], "annotations": []}

            if strand == '+':
                start = pos - 1
                end = start + len(seq.replace(pam, ""))
            else:
                start = pos - 1 + 3
                end = start + len(seq.replace(pam, ""))

            result[gene]["annotations"].append({
                "type": "sgRNA",
                "start": start,
                "end": end,
                "strand": 1 if strand == "+" else -1,
                "id": f"{gene}_{i}",
                "mm0": row["mm0_count"],
                "seq": seq
            })

            table.append({
                "gene": gene,
                "id": f"{gene}_{i}",
                "sgRNA": seq,
                "start": start,
                "end": end,
                "strand": strand,
                "mm0": row["mm0_count"],
                "mm1": row["mm1_count"],
                "bulge": row["bulge_count"],
                "specificity": row["specificity"]
            })

    json.dump(result, open(out_json, "w"), indent=2)

    with open(out_csv, "w", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=table[0].keys())
        writer.writeheader()
        writer.writerows(table)


# ================== 主流程 ==================
def run_pipeline(fasta, genome_index, prefix, pam, mismatch):

    print("🚀 Step1: generate kmers")
    subprocess.run(
        f"python {GENERATE_KMERS} --pam {pam} --kmer-length 20 --prefix {prefix} {fasta} > {prefix}.csv",
        shell=True, check=True
    )

    print("🚀 Step2: filter sgRNA")
    filter_sgrna(f"{prefix}.csv", f"{prefix}.filtered.csv")

    print("🚀 Step3: guidescan enumerate")
    subprocess.run(
        f"{GUIDESCAN_BIN} enumerate -m {mismatch} --max-off-targets 10 "
        f"--mode complete --format csv -n 6 "
        f"--output {prefix}.filtered.csv.enumerate {genome_index} "
        f"--kmers-file {prefix}.filtered.csv",
        shell=True, check=True
    )

    print("🚀 Step4: summarize")
    summarize_offtarget(
        f"{prefix}.filtered.csv.enumerate",
        f"{prefix}.final.csv"
    )

    print("🚀 Step5: build outputs")
    build_outputs(
        fasta,
        f"{prefix}.final.csv",
        f"{prefix}.json",
        f"{prefix}.table.csv",
        pam
    )

    print("✅ ALL DONE")


# ================== CLI ==================
if __name__ == "__main__":
    if len(sys.argv) != 6:
        print("Usage: python sgrna_pipeline.py input.fasta genome.index prefix pam mismatch")

def query_sgrna(conn, chrom, start, end):
    sql = """
    SELECT id, sgRNA, mm0_count, mm1_count, specificity, Chr, Pos, direction
    FROM sgrna
    WHERE Chr = ?
      AND Pos BETWEEN ? AND ?
    ORDER BY Pos
    """
    return pd.read_sql_query(sql, conn, params=(chrom, start, end))

def read_fasta_multi(fasta):
    seqs = {}
    current = None
    with open(fasta) as f:
        for line in f:
            if line.startswith(">"):
                current = line[1:].split()[0]
                seqs[current] = ""
            else:
                seqs[current] += line.strip()
    return seqs

def build_json(genome_name):
    DB_PATH = "../{}.db".format(genome_name)
    seqs = read_fasta_multi(INPUT_FASTA)
    df = pd.read_csv(INPUT_FILE, sep='\t')
    conn = sqlite3.connect(DB_PATH)
    results = {}
    Alltable = []
    for _, row in df.iterrows():
        chrom = row['Chr']
        gene = row['Gene']
        up = int(row['UpStart'])
        down = int(row['EndStart'])
        SeqName = row['SeqName']
        # ===== 查询 sgRNA =====
        sgrna_df = query_sgrna(conn, chrom, up, down)
        # ===== 解析 CDS =====
        cds_list = ast.literal_eval(row['CDSInfo'])
        annotations = []
        table = []

            cds_chr, cds_start, cds_end, cds_direction = cds
            annotations.append({
                "type": "CDS",
                "start": cds_start - up - 1,
                "end": cds_end - up,
                'strand': 1 if cds_direction == '+' else -1,
                "id": '{}-CDS'.format(gene),
                "mm0": '0',
                "specificity": 1,
                "seq": 'seq',
            })
        # ===== sgRNA 注释 =====
        for _, sg in sgrna_df.iterrows():
            pos = int(sg['Pos'])
            seq = sg['sgRNA']
            strand = 1 if sg['direction'] == '+' else -1
            if strand == 1:
                sgrna_start = pos - up - 1
                sgrna_end = sgrna_start + 20
            else:
                sgrna_start = pos - up + 2
                sgrna_end = sgrna_start + 20

            pam_end = pam_start + 3

import os.path

import pandas as pd

name = pd.DataFrame(pd.read_csv('name', sep='\t', header=None))

def analysis(name):
    anchors_file = "{}.lifted.anchors".format(name)
    anchors_df = pd.read_csv(anchors_file, sep=r"\s+", comment="#", header=None, names=["gene1", "gene2", "synteny_score"])
    anchor_rows = []
    for _, row in anchors_df.iterrows():
        g1, g2, score = row["gene1"], row["gene2"], row["synteny_score"]
        anchor_rows.append({
            "query_gene": g1,
            "hit_gene": g2,
            "method": "synteny",
            "score": score,
            "identity": '-',
            "aln_len": '-'
        })
    anchor_result_df = pd.DataFrame(anchor_rows)
    last_file = "{}.last.filtered".format(name)
    if os.path.exists(last_file) & (not os.path.exists("{}.gene_conversion.tsv".format(name))) :
        cols = ["query", "target", "identity", "aln_len", "mismatch", "gapopen", "q_start", "q_end", "t_start", "t_end", "strand", "bit_score"]
        last_df = pd.read_csv(last_file, sep=r"\s+", header=None, names=cols)
        last_df["blast_score"] = last_df["identity"] * last_df["aln_len"]
        last_result_df = (
            last_df
            .sort_values("blast_score", ascending=False)
            .drop_duplicates("query")
            .assign(
                query_gene=lambda x: x["query"],
                hit_gene=lambda x: x["target"],
                method="blast",
                score=lambda x: x["blast_score"]
            )[["query_gene", "hit_gene", "method", "score", "identity", "aln_len"]]
        )
        # anchors 里的 gene 集合
        anchor_genes = set(anchor_result_df["query_gene"])
        # 只保留 anchors 没覆盖到的 blast
        last_filtered_df = last_result_df[~last_result_df["query_gene"].isin(anchor_genes)]
        # 合并
        final_result_df = pd.concat([anchor_result_df, last_filtered_df], ignore_index=True)
        final_result_df = final_result_df.drop_duplicates("query_gene")
        final_result_df = final_result_df.set_index("query_gene")
        # 保存
        final_result_df = final_result_df.sort_values("query_gene")
        final_result_df.to_csv("{}.gene_conversion.tsv".format(name),sep="\t")
